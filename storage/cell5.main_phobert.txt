import torch
from torch.utils.data import Dataset
from transformers import AutoModelForSequenceClassification, Trainer, TrainingArguments
from sklearn.metrics import accuracy_score, f1_score
import numpy as np

# 1. Kế thừa và định nghĩa lớp Dataset chuẩn PyTorch cho bài toán NLP
class FeedbackDataset(Dataset):
    def __init__(self, encodings, labels):
        self.encodings = encodings
        self.labels = labels

    def __getitem__(self, idx):
        # Biến đổi dữ liệu sang định dạng tensor độc lập cho từng batch
        item = {key: val[idx].clone().detach() for key, val in self.encodings.items()}
        item['labels'] = torch.tensor(self.labels[idx], dtype=torch.long)
        return item

    def __len__(self):
        return len(self.labels)

# Khởi tạo đối tượng Dataset cho 2 tập dữ liệu
train_dataset = FeedbackDataset(train_encodings, y_train)
val_dataset = FeedbackDataset(val_encodings, y_val)

# 2. Phân tích số lượng nhãn phân loại (Number of Classes) từ dữ liệu gốc
# Ví dụ: Tích cực, Tiêu cực, Trung tính -> 3 nhãn
unique_labels = set(y_train)
num_labels = len(unique_labels)
print(f"📊 [PHÂN TÍCH DỮ LIỆU] Bài toán phân loại {num_labels} nhãn (Classes): {unique_labels}")

# 3. Tải kiến trúc mạng nơ-ron PhoBERT tinh chỉnh cho Phân loại chuỗi (Sequence Classification)
print("\n⏳ Đang khởi tạo kiến trúc mạng nơ-ron PhoBERT (AutoModelForSequenceClassification)...")
# Khởi tạo model với cấu trúc Classifier ở lớp layer cuối cùng (tương ứng với num_labels)
model = AutoModelForSequenceClassification.from_pretrained("vinai/phobert-base-v2", num_labels=num_labels)
print("✅ Tải kiến trúc mô hình thành công!")

# 4. Định nghĩa hàm tính toán các độ đo Đánh giá chất lượng mô hình (Evaluation Metrics)
def compute_metrics(pred):
    labels = pred.label_ids
    preds = pred.predictions.argmax(-1)
    
    # Tính toán Accuracy (Độ chính xác tổng thể) và F1-Macro (Độ chính xác cân bằng)
    acc = accuracy_score(labels, preds)
    f1 = f1_score(labels, preds, average='macro')
    return {
        'accuracy': acc,
        'f1_macro': f1
    }

# 5. Khai báo các Siêu tham số Huấn luyện (Hyperparameters)
# Các tham số này quyết định tốc độ học và khả năng chống học vẹt (overfitting) của AI
training_args = TrainingArguments(
    output_dir='./phobert_results',          # Thư mục lưu trữ trọng số mô hình
    num_train_epochs=4,                      # Số vòng lặp huấn luyện toàn bộ dữ liệu (Epochs)
    per_device_train_batch_size=16,          # Kích thước khối dữ liệu huấn luyện (Batch size)
    per_device_eval_batch_size=32,           # Kích thước khối dữ liệu đánh giá
    learning_rate=2e-5,                      # Tốc độ học (Learning Rate chuẩn cho Fine-tuning BERT)
    warmup_steps=500,                        # Khởi động chậm để tránh dao động loss giai đoạn đầu
    weight_decay=0.01,                       # Kỹ thuật phạt trọng số (L2 Regularization)
    eval_strategy="epoch",                   # Đánh giá sau mỗi epoch
    save_strategy="epoch",                   # Lưu trữ checkpoint sau mỗi epoch
    logging_dir='./logs',                    # Thư mục chứa log để theo dõi biểu đồ
    logging_steps=50,                        # Tần suất in log (cứ 50 steps in 1 lần)
    load_best_model_at_end=True,             # Tự động chọn mô hình có điểm đánh giá cao nhất ở cuối
    metric_for_best_model="f1_macro",        # Tiêu chí chọn mô hình tốt nhất là dựa vào F1-Score
)

# 6. Khởi tạo hệ thống Huấn luyện tự động (Trainer)
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=train_dataset,
    eval_dataset=val_dataset,
    compute_metrics=compute_metrics
)
